### 文字排版（全中文）

| 层级 | 字号/字重 | 颜色 | 示例 |
|------|----------|------|------|
| 总足迹数字 | 28px / 700 | `#fff` | **12** |
| 省份/城市数字 | 22px / 600 | `#fff` | **8 / 34** |
| 数据标签 | 11px / 500 | `rgba(255,255,255,0.60)` | 总足迹 |
| 分隔线 | 1px | `rgba(255,255,255,0.10)` | ───── |
| 分类行标签 | 12px / 400 | `rgba(255,255,255,0.55)` | 直辖市 |
| 分类行数字 | 12px / 500 | `#fff` | 2/4 |
| 去过=0 的行 | 整体 | `rgba(255,255,255,0.25)` | 淡出 |

### 省份颜色强调

省份数字 `8 / 34` 中的 `8` 使用主色调：
```css
color: hsl(var(--hsla-raw));
```
`/ 34` 保持白色 60%，形成"已去 / 总数"的视觉对比。

## 数据计算（纯前端 JS，不变）

```javascript
function computeStats(footprints) {
  const adcodeSet = new Set();
  const citySet = new Set();
  const muniPx = ['11','12','31','50'];
  const autoPx = ['15','45','54','64','65'];
  const sarPx  = ['81','82'];
  let m = 0, a = 0, s = 0, p = 0;

  footprints.forEach(fp => {
    const adc = fp.spec.provinceAdcode;
    if (!adc) return;
    const px = adc.substring(0, 2);
    if (!adcodeSet.has(adc)) {
      adcodeSet.add(adc);
      if (muniPx.includes(px)) m++;
      else if (autoPx.includes(px)) a++;
      else if (sarPx.includes(px)) s++;
      else p++;
    }
    if (fp.spec.cityAdcode) citySet.add(fp.spec.cityAdcode);
  });

  return {
    total: footprints.length,
    provinces: adcodeSet.size,
    cities: citySet.size,
    categories: [
      { label: '直辖市', v: m, t: 4 },
      { label: '自治区', v: a, t: 5 },
      { label: '特区',   v: s, t: 2 },
      { label: '省份',   v: p, t: 23 },
    ]
  };
}
```
